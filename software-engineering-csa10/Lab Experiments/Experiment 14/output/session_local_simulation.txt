# LOCAL SIMULATION - local bare repos stand in for GitHub (origin / upstream / fork)

## 01. Fork the repository (GitHub: Fork button)
# On GitHub: open the original repo -> click Fork -> Create fork. GitHub makes your own copy under your account.
# Equivalent of that copy, done here with git:
$ cd ~/git-lab
$ git clone --bare <original-repo-url> <your-fork-url>   # what the Fork button does
Cloning into bare repository 'remote/ashira-susan/open-source-demo.git'...
done.

## 02. Clone YOUR fork and add the upstream remote
$ cd ~/git-lab
$ git clone <your-fork-url>
Cloning into 'open-source-demo'...
done.
$ cd ~/git-lab/open-source-demo
$ git remote -v
origin	~/git-lab/remote/ashira-susan/open-source-demo.git (fetch)
origin	~/git-lab/remote/ashira-susan/open-source-demo.git (push)
$ git remote add upstream <original-repo-url>
$ git remote -v
origin	~/git-lab/remote/ashira-susan/open-source-demo.git (fetch)
origin	~/git-lab/remote/ashira-susan/open-source-demo.git (push)
upstream	~/git-lab/remote/course-instructor/open-source-demo.git (fetch)
upstream	~/git-lab/remote/course-instructor/open-source-demo.git (push)

## 03. Create a branch and make your contribution
$ cd ~/git-lab/open-source-demo
$ git switch -c add-susan-profile
Switched to a new branch 'add-susan-profile'
# add one row to CONTRIBUTORS.md and create contributors/susan.md
$ cat CONTRIBUTORS.md
# Contributors

| Name | GitHub | Profile |
|------|--------|---------|
| Course Instructor | @course-instructor | contributors/instructor.md |
| Susan | @ashira-susan | contributors/susan.md |
$ git status -s
 M CONTRIBUTORS.md
?? contributors/susan.md
$ git add contributors/susan.md CONTRIBUTORS.md
$ git commit -m "Add Susan to contributors"
[add-susan-profile 047a67a] Add Susan to contributors
 2 files changed, 5 insertions(+)
 create mode 100644 contributors/susan.md

## 04. Push the branch to your fork
$ cd ~/git-lab/open-source-demo
$ git push -u origin add-susan-profile
branch 'add-susan-profile' set up to track 'origin/add-susan-profile'.
To ~/git-lab/remote/ashira-susan/open-source-demo.git
 * [new branch]      add-susan-profile -> add-susan-profile
$ git branch -a
* add-susan-profile
  main
  remotes/origin/HEAD -> origin/main
  remotes/origin/add-susan-profile
  remotes/origin/main

## 05. Open the pull request (GitHub)
# GitHub shows 'Compare & pull request'. Base repository: original repo, base: main  <-  head repository: your fork, compare: add-susan-profile.
$ cd ~/git-lab/open-source-demo
$ git log main..add-susan-profile --oneline
047a67a Add Susan to contributors
$ git diff main...add-susan-profile --stat
 CONTRIBUTORS.md       | 1 +
 contributors/susan.md | 4 ++++
 2 files changed, 5 insertions(+)

## 06. Reviewer asks for changes - update the same PR
# Review comment: 'Please add a Skills section to your profile.'  Commit to the SAME branch and push - the PR updates automatically.
$ cd ~/git-lab/open-source-demo
$ git commit -am "Address review: add skills section"
[add-susan-profile b5a7cf2] Address review: add skills section
 1 file changed, 4 insertions(+)
$ git push
To ~/git-lab/remote/ashira-susan/open-source-demo.git
   047a67a..b5a7cf2  add-susan-profile -> add-susan-profile
$ git log main..add-susan-profile --oneline
b5a7cf2 Address review: add skills section
047a67a Add Susan to contributors

## 07. Maintainer merges the pull request
# Maintainer's computer (simulates the green 'Merge pull request' button).
$ cd ~/git-lab/maintainer
$ git remote add susan-fork <contributor-fork-url>
$ git fetch susan-fork
